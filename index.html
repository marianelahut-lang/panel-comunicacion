LEER PUBLICACIONES DE FACEBOOK (sin aprobación especial) // ══════════════════════════════════════════════════════════ async function fetchFBPosts(pageId, label) { if (!REDES_CONFIG.FB_APP_TOKEN || !pageId) { return { posts: [], error: "Sin configurar" }; } try { const url = `https://graph.facebook.com/v19.0/${pageId}/posts` + `?fields=message,created_time,full_picture,permalink_url,likes.summary(true),comments.summary(true),shares` + `&limit=20&access_token=${REDES_CONFIG.FB_APP_TOKEN}`; const r = await fetch(url); if (!r.ok) throw new Error("HTTP " + r.status); const d = await r.json(); if (d.error) throw new Error(d.error.message); return { posts: (d.data || []).map(p => ({ id: p.id, red: "Facebook", cuenta: label, texto: p.message || "(sin texto)", fecha: p.created_time?.slice(0, 10) || isoToday, hora: p.created_time?.slice(11, 16) || "", imagen: p.full_picture || null, url: p.permalink_url || "", likes: p.likes?.summary?.total_count || 0, comentarios: p.comments?.summary?.total_count || 0, compartidos: p.shares?.count || 0, })), error: null }; } catch(e) { console.warn("fetchFBPosts:", e.message); return { posts: [], error: e.message }; } } // ══════════════════════════════════════════════════════════ // 2.
